\section{文档配置}

编辑\texttt{main.tex}文件，可以设置学位类型、标题和作者信息等：

\begin{itemize}
    \item 学位类型：硕士论文使用\texttt{master}，博士论文使用\texttt{doctor}；
    \item 论文语言：按需要下载中文包或英文包，每个包都使用\texttt{main.tex}。
\end{itemize}

封面、中英文扉页和声明页按以下步骤导入：

\begin{enumerate}
    \item 下载与本人学位类型对应的Word模板：\href{https://gr.uestc.edu.cn/xiazai/114/3917}{https://gr.uestc.edu.cn/xiazai/114/3917}
    \item 填写本人的学位论文信息；
    \item 将封面及中英文扉页的Word文档导出为\texttt{cover.pdf}，然后覆盖项目中的同名文件。
        如需替换“独创性声明和论文使用授权”页，可将签署后的页面扫描为\texttt{declaration.pdf}，
        覆盖项目中的同名文件。
\end{enumerate}

\begin{equation}
    \mathbf{y}=\mathbf{A}\mathbf{x}.
    \label{eq:matrix-vector-product}

\section{定理与证明}

\begin{definition}
若矩阵$\mathbf{A}$满足$\mathbf{A}^{\mathsf{T}}\mathbf{A}=\mathbf{I}$，
则称$\mathbf{A}$为正交矩阵。
\end{definition}

\begin{theorem}
正交矩阵保持向量的二范数不变。
\end{theorem}

\begin{proof}
由$\mathbf{A}^{\mathsf{T}}\mathbf{A}=\mathbf{I}$可得
$\lVert\mathbf{A}\mathbf{x}\rVert_2^2
=\mathbf{x}^{\mathsf{T}}\mathbf{A}^{\mathsf{T}}\mathbf{A}\mathbf{x}
=\mathbf{x}^{\mathsf{T}}\mathbf{x}
=\lVert\mathbf{x}\rVert_2^2$。
\end{proof}
