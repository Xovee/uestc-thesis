\section{Document Configuration}

Edit \texttt{main.tex} to set the degree type, title, author information, and other options:

\begin{itemize}
    \item Degree type: use \texttt{master} for a master's thesis and \texttt{doctor} for a doctoral dissertation;
    \item Thesis language: download the Chinese or English package; each package uses \texttt{main.tex}.
\end{itemize}

Import the cover, Chinese and English title pages, and declaration page as follows:

\begin{enumerate}
    \item Download the appropriate Word template:
        \href{https://gr.uestc.edu.cn/xiazai/114/3917}{https://gr.uestc.edu.cn/xiazai/114/3917}
    \item Enter the thesis information;
    \item Export the cover and Chinese and English title pages as \texttt{cover.pdf}, then replace the file
        of the same name in the project. To replace the declaration of originality and authorization page,
        scan the signed page as \texttt{declaration.pdf} and replace the corresponding project file.
\end{enumerate}

\section{Theorems and Proofs}

\begin{definition}
A matrix $\mathbf{A}$ is orthogonal if $\mathbf{A}^{\mathsf{T}}\mathbf{A}=\mathbf{I}$.
\end{definition}

\begin{theorem}
An orthogonal matrix preserves the Euclidean norm of a vector.
\end{theorem}

\begin{proof}
Since $\mathbf{A}^{\mathsf{T}}\mathbf{A}=\mathbf{I}$,
$\lVert\mathbf{A}\mathbf{x}\rVert_2^2
=\mathbf{x}^{\mathsf{T}}\mathbf{A}^{\mathsf{T}}\mathbf{A}\mathbf{x}
=\mathbf{x}^{\mathsf{T}}\mathbf{x}
=\lVert\mathbf{x}\rVert_2^2$.
\end{proof}
